    10	export interface BandScaleCommandResult {

    18	  /**
    19	   * Helper to fetch instrument name maps.
    20	   */
    21	  private static async getInstrumentNames(orgId: string): Promise<Map<string, string>> {
    22	    const instrumentMap = new Map<string, string>();
    23	    if (!db) return instrumentMap;
    24	
    25	    try {
    26	      const snap = await db.collection("instruments").where("organizationId", "==", orgId).get();
    27	      for (const doc of snap.docs) {
    28	        instrumentMap.set(doc.id, doc.data().name);
    29	      }
    30	    } catch (e) {
    31	      logger.warn(`Failed to fetch instrument names for org ${orgId}: ${e}`);
    32	    }
    33	    return instrumentMap;
    34	  }
    35	
    36	  /**
    37	   * Helper to validate that all users exist and belong to the same organization.
    38	   */
    39	  private static async validateUsersOrganization(userIds: string[], orgId: string): Promise<void> {
    40	    if (!db || userIds.length === 0) return;
    41	
    42	    const uniqueUserIds = Array.from(new Set(userIds));
    43	    const userSnaps = await Promise.all(uniqueUserIds.map((id) => db.collection("users").doc(id).get()));
    44	
    45	    for (const snap of userSnaps) {
    46	      if (!snap.exists) {
    47	        throw new Error(`Integrante com ID ${snap.id} não foi encontrado no sistema.`);
    48	      }
    49	      const uData = snap.data();
    50	      if (uData?.organizationId !== orgId) {
    51	        throw new Error(`O integrante ${uData?.displayName || snap.id} não pertence a esta organização.`);
    52	      }
    53	    }
    54	  }

    56	  /**
    57	   * Helper to resolve author profile details.
    58	   */
    59	  private static async getAuthorProfile(userId: string): Promise<{ displayName: string; photoURL: string | null }> {
    60	    const defaultProfile = { displayName: "Líder de Escala", photoURL: null };
    61	    if (!db) return defaultProfile;
    62	
    63	    try {
    64	      const snap = await db.collection("users").doc(userId).get();
    65	      if (snap.exists) {
    66	        const data = snap.data() || {};
    67	        return {
    68	          displayName: data.displayName || "Líder de Escala",
    69	          photoURL: data.photoURL || null,
    70	        };
    71	      }
    72	    } catch (e) {}
    73	    return defaultProfile;
    74	  }

    76	  /**
    77	   * Create a brand new Band Scale with responses and notifications inside an atomic transaction.
    78	   */
    79	  static async createBandScale(params: {
    80	    authUid: string;
    81	    orgId: string;
    82	    idempotencyKey: string;
    83	    payload: any;
    84	    correlationId: string;
    85	  }): Promise<BandScaleCommandResult> {
    86	    const startTime = Date.now();
    87	    const { authUid, orgId, idempotencyKey, payload, correlationId } = params;
    88	    const commandId = crypto.randomUUID();
    89	
    90	    // 1. Validate Idempotency Key Format
    91	    if (!idempotencyKey || typeof idempotencyKey !== "string" || idempotencyKey.trim().length < 10) {
    92	      throw new Error("Chave de idempotência ausente ou com formato inválido.");
    93	    }
    94	
    95	    const receiptId = IdempotencyService.getReceiptId(orgId, idempotencyKey);
    96	    const fingerprint = IdempotencyService.getRequestFingerprint(payload);
    97	
    98	    if (!db) {
    99	      throw new Error("Banco de dados não disponível.");
   100	    }
   101	
   102	    // 2. Authorization (RBAC)
   103	    const isAuthorized = await BandScaleAuthorizationService.checkCanManageScales(authUid, orgId);
   104	    if (!isAuthorized) {
   105	      throw new Error("Permissão canManageScales negada para esta operação.");
   106	    }
   107	
   108	    // 3. Normalize & Validate Assignments
   109	    const rawAssignments = payload.assignments || [];
   110	    const scaleId = crypto
   111	      .createHash("sha256")
   112	      .update(`scale:${orgId}:${idempotencyKey}`)
   113	      .digest("hex")
   114	      .substring(0, 20);
   115	
   116	    const reconciledAssignments = AssignmentNormalizer.reconcile([], rawAssignments, scaleId);
   117	
   118	    // 4. Validate Users and fetch resources (parallelized pre-reads)
   119	    const userIds = reconciledAssignments.map((a) => a.userId);
   120	    const [authorProfile, instrumentNames] = await Promise.all([
   121	      this.getAuthorProfile(authUid),
   122	      this.getInstrumentNames(orgId),
   123	      this.validateUsersOrganization(userIds, orgId),
   124	    ]);
   125	
   126	    // 5. Estimate transactional write volume
   127	    // Writes: 1 BandScale + N Responses + N Notifications + 1 Receipt
   128	    const totalWrites = 1 + reconciledAssignments.length + reconciledAssignments.length + 1;
   129	    if (totalWrites > 400) {
   130	      throw new Error("Payload muito grande. O volume de operações excede o limite seguro para uma transação.");
   131	    }
   132	
   133	    // 6. Run Atomic Transaction
   134	    const result = await db.runTransaction(async (transaction) => {
   135	      // Check existing idempotency receipt
   136	      const existingReceipt = await IdempotencyService.getReceiptInTransaction(transaction, orgId, receiptId);
   137	      if (existingReceipt) {
   138	        if (existingReceipt.requestFingerprint !== fingerprint) {
   139	          throw new Error("Esta chave de idempotência já foi utilizada com um payload diferente.");
   140	        }
   141	        return {
   142	          scaleId: existingReceipt.result.scaleId,
   143	          version: existingReceipt.result.version,
   144	          createdNotificationCount: existingReceipt.result.createdNotificationCount,
   145	          fromCache: true,
   146	        };
   147	      }
   148	
   149	      const bandScaleRef = db.collection("bandScales").doc(scaleId);
   150	
   151	      // Construct scale
   152	      const bandScaleDoc = {
   153	        id: scaleId,
   154	        organizationId: orgId,
   155	        date: payload.date || null,
   156	        time: payload.time || null,
   157	        observations: payload.observations || null,
   158	        eventTypeId: payload.eventTypeId || null,
   159	        locationId: payload.locationId || null,
   160	        eventNameId: payload.eventNameId || null,
   161	        musicScaleId: payload.musicScaleId || null,
   162	        version: 1,
   163	        createdAt: admin.firestore.FieldValue.serverTimestamp(),
   164	        createdBy: authorProfile,
   165	        updatedAt: admin.firestore.FieldValue.serverTimestamp(),
   166	        updatedBy: authUid,
   167	        notificationProducer: "command_api",
   168	        assignments: reconciledAssignments,
   169	      };
   170	
   171	      transaction.set(bandScaleRef, bandScaleDoc);
   172	
   173	      const notificationCount = 0;
   174	
   175	      // Write Idempotency Receipt
   176	      IdempotencyService.writeReceiptInTransaction(transaction, orgId, receiptId, {
   177	        commandType: "bandScale.create",
   178	        organizationId: orgId,
   179	        userId: authUid,
   180	        entityId: scaleId,
   181	        requestFingerprint: fingerprint,
   182	        result: {
   183	          scaleId,
   184	          version: 1,
   185	          createdNotificationCount: notificationCount,
   186	        },
   187	      });
   188	
   189	      return {
   190	        scaleId,
   191	        version: 1,
   192	        createdNotificationCount: notificationCount,
   193	        fromCache: false,
   194	      };
   195	    });
   196	
   197	    const durationMs = Date.now() - startTime;
   198	    logger.info("BandScale creation completed", {
   199	      correlationId,
   200	      commandId,
   201	      organizationId: orgId,
   202	      authenticatedUserId: authUid,
   203	      scaleId,
   204	      action: "create",
   205	      previousVersion: 0,
   206	      newVersion: result.version,
   207	      assignmentCount: reconciledAssignments.length,
   208	      createdResponseCount: reconciledAssignments.length,
   209	      createdNotificationCount: result.createdNotificationCount,
   210	      durationMs,
   211	      fromCache: !!(result as any).fromCache,
   212	    });
   213	
   214	    return {
   215	      scaleId: result.scaleId,
   216	      version: result.version,
   217	      createdNotificationCount: result.createdNotificationCount,
   218	      correlationId,
   219	    };
   220	  }
